// ===== PROCESS CARD PAYMENT ===== function processCardPayment(e) { e.preventDefault(); const cardNumber = document.getElementById('cardNumber').value.replace(/\s/g, ''); const expiry = document.getElementById('expiry').value; const cvv = document.getElementById('cvv').value; const cardholder = document.getElementById('cardholder').value.trim(); const payBtn = document.getElementById('payBtn'); let isValid = true; // Validate Card Number if (cardNumber.length < 16) { document.getElementById('cardError').classList.add('show'); isValid = false; } // Validate Expiry if (!/^\d{2}\/\d{2}$/.test(expiry)) { document.getElementById('expiryError').classList.add('show'); isValid = false; } else { const [month, year] = expiry.split('/'); const now = new Date(); const expYear = 2000 + parseInt(year); const expMonth = parseInt(month) - 1; const expDate = new Date(expYear, expMonth, 1); if (expDate < now) { document.getElementById('expiryError').textContent = 'This card has expired'; document.getElementById('expiryError').classList.add('show'); isValid = false; } } // Validate CVV if (cvv.length < 3 || cvv.length > 4) { document.getElementById('cvvError').classList.add('show'); isValid = false; } // Validate Cardholder Name if (cardholder.length < 2) { document.getElementById('nameError').classList.add('show'); isValid = false; } if (!isValid) { return; } // Get current user const currentUser = JSON.parse(localStorage.getItem('currentUser')); // Generate order ID const orderId = 'OZ-' + Date.now().toString().slice(-6) + Math.floor(Math.random() * 1000); // Prepare payment data const paymentData = { orderId: orderId, customerName: currentUser ? currentUser.name : 'Unknown', customerEmail: currentUser ? currentUser.email || 'N/A' : 'N/A', customerPhone: currentUser ? currentUser.phone || 'N/A' : 'N/A', paymentMethod: 'card', cardLastFour: cardNumber.slice(-4), cardType: detectCardType(cardNumber) || 'Unknown', cardholderName: cardholder, amount: 'R 150.00', status: 'completed', date: new Date().toISOString(), timestamp: firebase.database.ServerValue.TIMESTAMP }; // Show processing state payBtn.innerHTML = ' Processing...'; payBtn.disabled = true; // Save to Firebase const paymentsRef = firebase.database().ref('payments'); const newPaymentRef = paymentsRef.push(); newPaymentRef.set(paymentData) .then(() => { console.log('Card payment saved to Firebase:', orderId); localStorage.setItem('orderId', orderId); localStorage.setItem('paymentStatus', 'success'); localStorage.setItem('paymentAmount', 'R 150.00'); localStorage.setItem('cardLastFour', cardNumber.slice(-4)); localStorage.setItem('cardholderName', cardholder); // Redirect to confirmation setTimeout(() => { window.location.href = 'cash-confirmation.html'; }, 500); }) .catch((error) => { console.error('Error saving payment:', error); alert('There was an error processing your payment. Please try again.'); payBtn.innerHTML = ' Pay Now'; payBtn.disabled = false; }); }
